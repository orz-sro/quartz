\qquad  \large =\int \frac{1}{2}\, x \,\sec ^2 \frac{x}{2}+\tan \frac{x}{2}\,dx   \\ \\ 
 \large =\int d\left( x\tan \frac{x}{2} \right) \\ \\
\large I=x\tan \frac{x}{2}+C
\end{gather}
